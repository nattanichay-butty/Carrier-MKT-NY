    /**
     * Client-side Image Compression & Base64 Reader
     * Automatically scales images down to max 1280px to ensure fast upload and lean payloads
     */
    function compressAndReadImage(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = (event) => {
          const img = new Image();
          img.src = event.target.result;
          img.onload = () => {
            const canvas = document.createElement('canvas');
            const maxDimension = 1280;
            let width = img.width;
            let height = img.height;

            if (width > height) {
              if (width > maxDimension) {
                height = Math.round((height * maxDimension) / width);
                width = maxDimension;
              }
            } else {
              if (height > maxDimension) {
                width = Math.round((width * maxDimension) / height);
                height = maxDimension;
              }
            }

            canvas.width = width;
            canvas.height = height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, width, height);

            // Compress to JPEG with 0.82 quality
            const compressedBase64 = canvas.toDataURL('image/jpeg', 0.82);
            resolve(compressedBase64);
          };
          img.onerror = (err) => reject(err);
        };
        reader.onerror = (err) => reject(err);
      });
    }

    // Live Diagnostic: Check if URL allows unauthenticated access (GET Ping)
    async function runDiagnostics(url) {
      if (!url) {
        return { ok: false, error: "ยังไม่ได้ระบุ Web App URL" };
      }
      try {
        const response = await fetch(url + (url.includes('?') ? '&' : '?') + 'ping=' + Date.now(), {
          method: 'GET',
          redirect: 'follow',
          mode: 'cors'
        });
        
        if (response.ok) {
          const data = await response.json();
          return { ok: true, data: data };
        } else {
          return { ok: false, error: `ได้รับสถานะ HTTP ${response.status} จาก Google` };
        }
      } catch (err) {
        // In CORS error cases, usually blocked by Google Auth redirect

    function validateForm() {
      let isValid = true;

      // Store Name
      const storeName = document.getElementById('storeName');
      const storeNameError = document.getElementById('storeNameError');
      if (!storeName.value.trim()) {
        storeNameError.classList.remove('hidden');
        storeName.classList.add('border-rose-400', 'bg-rose-50/30');
        isValid = false;
      } else {
        storeNameError.classList.add('hidden');
        storeName.classList.remove('border-rose-400', 'bg-rose-50/30');
      }

      // Contact Name
      const contactName = document.getElementById('contactName');
      const contactNameError = document.getElementById('contactNameError');
      if (!contactName.value.trim()) {
        contactNameError.classList.remove('hidden');
        contactName.classList.add('border-rose-400', 'bg-rose-50/30');
        isValid = false;
      } else {
        contactNameError.classList.add('hidden');
        contactName.classList.remove('border-rose-400', 'bg-rose-50/30');
      }

      // Phone (9-10 digits, Thai formats)
      const phone = document.getElementById('phone');
      const phoneError = document.getElementById('phoneError');
      const phoneClean = phone.value.replace(/[^0-9]/g, '');
      if (phoneClean.length < 9 || phoneClean.length > 10) {
        phoneError.classList.remove('hidden');
        phone.classList.add('border-rose-400', 'bg-rose-50/30');
        isValid = false;
      } else {
        phoneError.classList.add('hidden');
        phone.classList.remove('border-rose-400', 'bg-rose-50/30');
      }

      // Address
      const address = document.getElementById('address');
      const addressError = document.getElementById('addressError');
      if (!address.value.trim()) {
        addressError.classList.remove('hidden');
        address.classList.add('border-rose-400', 'bg-rose-50/30');
        isValid = false;
      } else {
        addressError.classList.add('hidden');
        address.classList.remove('border-rose-400', 'bg-rose-50/30');
      }

      // Location fields (subdistrict, district, province, postalCode)
      const subdistrict = document.getElementById('subdistrict').value.trim();
      const district = document.getElementById('district').value.trim();
      const province = document.getElementById('province').value.trim();
      const postalCode = document.getElementById('postalCode').value.trim();
      const locationError = document.getElementById('locationError');

      if (!subdistrict || !district || !province || !postalCode) {
        locationError.classList.remove('hidden');
        isValid = false;
      } else {
        locationError.classList.add('hidden');
      }

      // Consent
      const consent = document.getElementById('consent');
      const consentError = document.getElementById('consentError');
      if (!consent.checked) {
        consentError.classList.remove('hidden');
        isValid = false;
      } else {
        consentError.classList.add('hidden');
      }

      return isValid;
    }

ขั้นตอนการติดตั้ง (ใช้เวลา 2 นาที):

  1. เปิด Google Sheets แผ่นใหม่ (sheets.new)
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดสำหรับร้านแอร์ด้านล่างนี้ไปวาง:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // สร้างหัวตารางอัตโนมัติหากเป็นชีทใหม่
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "วันที่-เวลา", 
        "ชื่อร้านค้า/บริษัทแอร์", 
        "ชื่อผู้ติดต่อ", 
        "เบอร์โทรติดต่อ", 
        "LINE ID", 
        "เลขประจำตัวผู้เสียภาษี", 
        "ที่อยู่/โกดัง", 
        "ตำบล/แขวง", 
        "อำเภอ/เขต", 
        "จังหวัด", 
        "รหัสไปรษณีย์", 
        "Google Maps",
        "ลักษณะบริการของร้าน", 
        "จำนวนทีมช่างติดตั้ง", 
        "ยอดจำหน่ายเฉลี่ย (เครื่อง/ด.)", 
        "แบรนด์แอร์ที่ขายปัจจุบัน", 
        "สินค้าที่สนใจสั่งซื้อ", 
        "หมายเหตุ/เงื่อนไขที่ต้องการ"
      ]);
      // ปรับแต่งสีหัวตาราง
      sheet.getRange(1, 1, 1, 18).setBackground("#0369a1").setFontColor("#FFFFFF").setFontWeight("bold");
    }
    
    // บันทึกแถวข้อมูลร้านแอร์ใหม่
    sheet.appendRow([
      new Date(),
      data.storeName,
      data.contactName,
      data.phone,
      data.lineId || "-",
      data.taxId || "-",
      data.address,
      data.subdistrict,
      data.district,
      data.province,
      data.postalCode,
      data.mapUrl || "-",
      data.businessType,
      data.technicianCount,
      data.monthlySales,
      (data.currentBrands || []).join(", "),
      (data.interestedProducts || []).join(", "),
      data.notes || "-"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

  2. เลือกประเภท เว็บแอป (Web app)
